   134→// ==================== DropdownWithManage Component ====================
   135→function DropdownWithManage({
   136→  label,
   137→  type,
   138→  value,
   139→  onChange,
   140→  disabled,
   141→}: {
   142→  label: string;
   143→  type: "department" | "branch" | "address";
   144→  value: string;
   145→  onChange: (val: string) => void;
   146→  disabled: boolean;
   147→}) {
   148→  const [options, setOptions] = useState<DropdownOption[]>([]);
   149→  const [manageOpen, setManageOpen] = useState(false);
   150→  const [newOptionValue, setNewOptionValue] = useState("");
   151→  const [editingId, setEditingId] = useState<string | null>(null);
   152→  const [editingValue, setEditingValue] = useState("");
   153→  const [loading, setLoading] = useState(false);
   154→  const { toast } = useToast();
   155→
   156→  const fetchOptions = useCallback(async () => {
   157→    try {
   158→      const res = await fetch(`/api/dropdown-options?type=${type}`);
   159→      if (res.ok) setOptions(await res.json());
   160→    } catch { /* ignore */ }
   161→  }, [type]);
   162→
   163→  useEffect(() => { fetchOptions(); }, [fetchOptions]);
   164→
   165→  const handleAdd = async () => {
   166→    if (!newOptionValue.trim()) return;
   167→    setLoading(true);
   168→    try {
   169→      const res = await fetch("/api/dropdown-options", {
   170→        method: "POST",
   171→        headers: { "Content-Type": "application/json" },
   172→        body: JSON.stringify({ type, value: newOptionValue.trim() }),
   173→      });
   174→      if (res.ok) {
   175→        toast({ title: "Added", description: `"${newOptionValue.trim()}" added to ${label}` });
   176→        setNewOptionValue("");
   177→        await fetchOptions();
   178→      } else {
   179→        const err = await res.json();
   180→        toast({ title: "Error", description: err.error || "Failed to add", variant: "destructive" });
   181→      }
   182→    } catch {
   183→      toast({ title: "Error", description: "Failed to add option", variant: "destructive" });
   184→    } finally { setLoading(false); }
   185→  };
   186→
   187→  const handleUpdate = async (id: string) => {
   188→    if (!editingValue.trim()) return;
   189→    setLoading(true);
   190→    try {
   191→      const res = await fetch(`/api/dropdown-options/${id}`, {
   192→        method: "PUT",
   193→        headers: { "Content-Type": "application/json" },
   194→        body: JSON.stringify({ value: editingValue.trim() }),
   195→      });
   196→      if (res.ok) {
   197→        toast({ title: "Updated", description: "Option updated successfully" });
   198→        setEditingId(null);
   199→        setEditingValue("");
   200→        await fetchOptions();
   201→      } else {
   202→        toast({ title: "Error", description: "Failed to update", variant: "destructive" });
   203→      }
   204→    } catch {
   205→      toast({ title: "Error", description: "Failed to update", variant: "destructive" });
   206→    } finally { setLoading(false); }
   207→  };
   208→
   209→  const handleDelete = async (id: string, val: string) => {
   210→    setLoading(true);
   211→    try {
   212→      const res = await fetch(`/api/dropdown-options/${id}`, { method: "DELETE" });
   213→      if (res.ok) {
   214→        toast({ title: "Deleted", description: `"${val}" removed from ${label}` });
   215→        if (value === val) onChange("");
   216→        await fetchOptions();
   217→      }
   218→    } catch {
   219→      toast({ title: "Error", description: "Failed to delete", variant: "destructive" });
   220→    } finally { setLoading(false); }
   221→  };
   222→
   223→  return (
   224→    <div className="flex items-center gap-1">
   225→      <Select value={value} onValueChange={onChange} disabled={disabled}>
   226→        <SelectTrigger className="h-7 text-xs flex-1">
   227→          <SelectValue placeholder={`Select ${label}...`} />
   228→        </SelectTrigger>
   229→        <SelectContent>
   230→          {options.map((opt) => (
   231→            <SelectItem key={opt.id} value={opt.value}>{opt.value}</SelectItem>
   232→          ))}
   233→        </SelectContent>
   234→      </Select>
   235→      {!disabled && (
   236→        <Popover open={manageOpen} onOpenChange={setManageOpen}>
   237→          <PopoverTrigger asChild>
   238→            <Button variant="ghost" size="sm" className="h-7 w-7 p-0 shrink-0 text-slate-400 hover:text-slate-700" title="Manage options">
   239→              <Settings2 className="w-3.5 h-3.5" />
   240→            </Button>
   241→          </PopoverTrigger>
   242→          <PopoverContent className="w-72 p-0" align="end">
   243→            <div className="p-3 border-b bg-slate-50">
   244→              <h4 className="text-sm font-semibold text-slate-700">Manage {label}</h4>
   245→              <p className="text-xs text-slate-400 mt-0.5">Add, edit or delete options</p>
   246→            </div>
   247→            <div className="p-3 border-b">
   248→              <div className="flex gap-1.5">
   249→                <Input
   250→                  placeholder={`New ${label.toLowerCase()}...`}
   251→                  value={newOptionValue}
   252→                  onChange={(e) => setNewOptionValue(e.target.value)}
   253→                  className="h-7 text-xs"
   254→                  onKeyDown={(e) => e.key === "Enter" && handleAdd()}
   255→                />
   256→                <Button size="sm" onClick={handleAdd} disabled={loading || !newOptionValue.trim()} className="h-7 px-2 shrink-0 bg-slate-700 hover:bg-slate-600">
   257→                  <Plus className="w-3.5 h-3.5" />
   258→                </Button>
   259→              </div>
   260→            </div>
   261→            <div className="max-h-48 overflow-y-auto">
   262→              {options.length === 0 ? (
   263→                <p className="p-3 text-xs text-slate-400 text-center">No options yet</p>
   264→              ) : (
   265→                options.map((opt) => (
   266→                  <div key={opt.id} className="flex items-center gap-1 px-3 py-1.5 border-b last:border-b-0 hover:bg-slate-50 group">
   267→                    {editingId === opt.id ? (
   268→                      <>
   269→                        <Input
   270→                          value={editingValue}
   271→                          onChange={(e) => setEditingValue(e.target.value)}
   272→                          className="h-6 text-xs flex-1"
   273→                          onKeyDown={(e) => e.key === "Enter" && handleUpdate(opt.id)}
   274→                          autoFocus
   275→                        />
   276→                        <Button size="sm" onClick={() => handleUpdate(opt.id)} disabled={loading} className="h-6 w-6 p-0 shrink-0 text-green-600 hover:text-green-700">
   277→                          <Check className="w-3 h-3" />
   278→                        </Button>
   279→                        <Button size="sm" variant="ghost" onClick={() => { setEditingId(null); setEditingValue(""); }} className="h-6 w-6 p-0 shrink-0">
   280→                          <X className="w-3 h-3" />
   281→                        </Button>
   282→                      </>
   283→                    ) : (
   284→                      <>
   285→                        <span className="text-xs flex-1 truncate">{opt.value}</span>
   286→                        <Button
   287→                          size="sm" variant="ghost"
   288→                          onClick={() => { setEditingId(opt.id); setEditingValue(opt.value); }}
   289→                          className="h-6 w-6 p-0 shrink-0 opacity-0 group-hover:opacity-100 text-blue-500 hover:text-blue-700"
   290→                        >
   291→                          <Pencil className="w-3 h-3" />
   292→                        </Button>
   293→                        <Button
   294→                          size="sm" variant="ghost"
   295→                          onClick={() => handleDelete(opt.id, opt.value)}
   296→                          disabled={loading}
   297→                          className="h-6 w-6 p-0 shrink-0 opacity-0 group-hover:opacity-100 text-red-500 hover:text-red-700"
   298→                        >
   299→                          <Trash2 className="w-3 h-3" />
   300→                        </Button>
   301→                      </>
   302→                    )}
   303→                  </div>
   304→                ))
   305→              )}
   306→            </div>
   307→          </PopoverContent>
   308→        </Popover>
   309→      )}
   310→    </div>
   311→  );
   312→}

   314→// ==================== DatePickerWithAutoClose Component ====================
   315→function DatePickerWithAutoClose({
   316→  value,
   317→  onChange,
   318→}: {
   319→  value: string;
   320→  onChange: (val: string) => void;
   321→}) {
   322→  const [open, setOpen] = useState(false);
   323→
   324→  const parseDate = (dateStr: string): Date | undefined => {
   325→    if (!dateStr) return undefined;
   326→    const parts = dateStr.split("/");
   327→    if (parts.length === 3) {
   328→      return new Date(Number(parts[2]), Number(parts[1]) - 1, Number(parts[0]));
   329→    }
   330→    return undefined;
   331→  };
   332→
   333→  return (
   334→    <Popover open={open} onOpenChange={setOpen}>
   335→      <PopoverTrigger asChild>
   336→        <Button
   337→          variant="outline"
   338→          className="mt-1 w-full justify-start text-left font-normal h-9"
   339→        >
   340→          <CalendarIcon className="mr-2 h-4 w-4 text-slate-500" />
   341→          {value || "Pick a date"}
   342→        </Button>
   343→      </PopoverTrigger>
   344→      <PopoverContent className="w-auto p-0" align="start">
   345→        <Calendar
   346→          mode="single"
   347→          selected={parseDate(value)}
   348→          onSelect={(date) => {
   349→            if (date) {
   350→              onChange(format(date, "dd/MM/yyyy"));
   351→              setOpen(false);
   352→            }
   353→          }}
   354→          captionLayout="dropdown-buttons"
   355→          defaultMonth={parseDate(value) || new Date()}
   356→        />
   357→      </PopoverContent>
   358→    </Popover>
   359→  );
   360→}
