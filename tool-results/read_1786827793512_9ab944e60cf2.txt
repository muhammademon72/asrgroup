   143→// ==================== DropdownWithManage Component ====================
   144→function DropdownWithManage({
   145→  label,
   146→  type,
   147→  value,
   148→  onChange,
   149→  disabled,
   150→}: {
   151→  label: string;
   152→  type: "department" | "branch" | "address";
   153→  value: string;
   154→  onChange: (val: string) => void;
   155→  disabled: boolean;
   156→  isAdmin: boolean;
   157→}) {
   158→  const [options, setOptions] = useState<DropdownOption[]>([]);
   159→
   160→  const fetchOptions = useCallback(async () => {
   161→    try {
   162→      const res = await fetch(`/api/dropdown-options?type=${type}`);
   163→      if (res.ok) setOptions(await res.json());
   164→    } catch { /* ignore */ }
   165→  }, [type]);
   166→
   167→  useEffect(() => { fetchOptions(); }, [fetchOptions]);
   168→
   169→  return (
   170→    <Select value={value} onValueChange={onChange} disabled={disabled}>
   171→      <SelectTrigger className="h-7 text-xs w-full">
   172→        <SelectValue placeholder={`Select ${label}...`} />
   173→      </SelectTrigger>
   174→      <SelectContent>
   175→        {/* Show current value as fallback if it doesn't match any DB option */}
   176→        {value && !options.some(opt => opt.value === value) && (
   177→          <SelectItem key="__current__" value={value}>{value}</SelectItem>
   178→        )}
   179→        {options.map((opt) => (
   180→          <SelectItem key={opt.id} value={opt.value}>{opt.value}</SelectItem>
   181→        ))}
   182→      </SelectContent>
   183→    </Select>
   184→  );

   187→// ==================== DatePickerWithAutoClose Component ====================
   188→function DatePickerWithAutoClose({
   189→  value,
   190→  onChange,
   191→}: {
   192→  value: string;
   193→  onChange: (val: string) => void;
   194→}) {
   195→  const [open, setOpen] = useState(false);
   196→
   197→  const parseDate = (dateStr: string): Date | undefined => {
   198→    if (!dateStr) return undefined;
   199→    const parts = dateStr.split("/");
   200→    if (parts.length === 3) {
   201→      return new Date(Number(parts[2]), Number(parts[1]) - 1, Number(parts[0]));
   202→    }
   203→    return undefined;
   204→  };
   205→
   206→  return (
   207→    <Popover open={open} onOpenChange={setOpen}>
   208→      <PopoverTrigger asChild>
   209→        <Button
   210→          variant="outline"
   211→          className="mt-1 w-full justify-start text-left font-normal h-9"
   212→        >
   213→          <CalendarIcon className="mr-2 h-4 w-4 text-slate-500" />
   214→          {value || "Pick a date"}
   215→        </Button>
   216→      </PopoverTrigger>
   217→      <PopoverContent className="w-auto p-0" align="start">
   218→        <Calendar
   219→          mode="single"
   220→          selected={parseDate(value)}
   221→          onSelect={(date) => {
   222→            if (date) {
   223→              onChange(format(date, "dd/MM/yyyy"));
   224→              setOpen(false);
   225→            }
   226→          }}
   227→          captionLayout="dropdown-buttons"
   228→          defaultMonth={parseDate(value) || new Date()}
   229→        />
   230→      </PopoverContent>
   231→    </Popover>
   232→  );
   233→}

   235→// ==================== Dropdown Management Component ====================
   236→function DropdownManagement({ type, label, icon }: { type: "department" | "branch" | "address"; label: string; icon: React.ReactNode }) {
   237→  const [options, setOptions] = useState<DropdownOption[]>([]);
   238→  const [newValue, setNewValue] = useState("");
   239→  const [editingId, setEditingId] = useState<string | null>(null);
   240→  const [editingValue, setEditingValue] = useState("");
   241→  const [deleteId, setDeleteId] = useState<string | null>(null);
   242→  const [deleteDialogOpen, setDeleteDialogOpen] = useState(false);
   243→  const [loading, setLoading] = useState(false);
   244→  const { toast } = useToast();
   245→
   246→  const fetchOptions = useCallback(async () => {
   247→    try {
   248→      const res = await fetch(`/api/dropdown-options?type=${type}`);
   249→      if (res.ok) setOptions(await res.json());
   250→    } catch { /* ignore */ }
   251→  }, [type]);
   252→
   253→  useEffect(() => { fetchOptions(); }, [fetchOptions]);
   254→
   255→  const handleAdd = async () => {
   256→    if (!newValue.trim()) return;
   257→    setLoading(true);
   258→    try {
   259→      const res = await fetch("/api/dropdown-options", {
   260→        method: "POST",
   261→        headers: { "Content-Type": "application/json" },
   262→        body: JSON.stringify({ type, value: newValue.trim() }),
   263→      });
   264→      if (res.ok) {
   265→        toast({ title: "Added", description: `"${newValue.trim()}" added successfully` });
   266→        setNewValue("");
   267→        await fetchOptions();
   268→      } else {
   269→        const err = await res.json();
   270→        toast({ title: "Error", description: err.error || "Failed to add", variant: "destructive" });
   271→      }
   272→    } catch {
   273→      toast({ title: "Error", description: "Failed to add option", variant: "destructive" });
   274→    } finally { setLoading(false); }
   275→  };
   276→
   277→  const handleUpdate = async () => {
   278→    if (!editingId || !editingValue.trim()) return;
   279→    setLoading(true);
   280→    try {
   281→      const res = await fetch(`/api/dropdown-options/${editingId}`, {
   282→        method: "PUT",
   283→        headers: { "Content-Type": "application/json" },
   284→        body: JSON.stringify({ value: editingValue.trim() }),
   285→      });
   286→      if (res.ok) {
   287→        toast({ title: "Updated", description: "Option updated successfully" });
   288→        setEditingId(null);
   289→        setEditingValue("");
   290→        await fetchOptions();
   291→      } else {
   292→        toast({ title: "Error", description: "Failed to update", variant: "destructive" });
   293→      }
   294→    } catch {
   295→      toast({ title: "Error", description: "Failed to update", variant: "destructive" });
   296→    } finally { setLoading(false); }
   297→  };
   298→
   299→  const handleDelete = async () => {
   300→    if (!deleteId) return;
   301→    setLoading(true);
   302→    try {
   303→      const res = await fetch(`/api/dropdown-options/${deleteId}`, { method: "DELETE" });
   304→      if (res.ok) {
   305→        toast({ title: "Deleted", description: "Option removed successfully" });
   306→        setDeleteId(null);
   307→        setDeleteDialogOpen(false);
   308→        await fetchOptions();
   309→      }
   310→    } catch {
   311→      toast({ title: "Error", description: "Failed to delete", variant: "destructive" });
   312→    } finally { setLoading(false); }
   313→  };
   314→
   315→  return (
   316→    <div className="space-y-6">
   317→      {/* Header */}
   318→      <div className="flex items-center gap-4">
   319→        <div className="w-12 h-12 rounded-xl bg-slate-800 flex items-center justify-center text-white">
   320→          {icon}
   321→        </div>
   322→        <div>
   323→          <h2 className="text-xl font-bold text-slate-800">Manage {label}</h2>
   324→          <p className="text-sm text-slate-500">Create, edit, and delete {label.toLowerCase()} options</p>
   325→        </div>
   326→      </div>
   327→
   328→      {/* Add New */}
   329→      <Card className="shadow-sm">
   330→        <CardContent className="p-4">
   331→          <div className="flex items-center gap-3">
   332→            <Input
   333→              placeholder={`Enter new ${label.toLowerCase()} name...`}
   334→              value={newValue}
   335→              onChange={(e) => setNewValue(e.target.value)}
   336→              onKeyDown={(e) => e.key === "Enter" && handleAdd()}
   337→              className="flex-1 h-9"
   338→            />
   339→            <Button onClick={handleAdd} disabled={loading || !newValue.trim()} className="bg-slate-800 hover:bg-slate-700 gap-2 h-9">
   340→              {loading ? <Loader2 className="w-4 h-4 animate-spin" /> : <Plus className="w-4 h-4" />} Add
   341→            </Button>
   342→          </div>
   343→        </CardContent>
   344→      </Card>
   345→
   346→      {/* Options List */}
   347→      <Card className="shadow-sm">
   348→        <div className="border-b bg-slate-50 px-4 py-3 flex items-center justify-between">
   349→          <h3 className="text-sm font-semibold text-slate-700">{label} List ({options.length})</h3>
   350→        </div>
   351→        {options.length === 0 ? (
   352→          <CardContent className="py-12 text-center">
   353→            <Settings2 className="w-12 h-12 mx-auto text-slate-300 mb-3" />
   354→            <p className="text-sm text-slate-500">No {label.toLowerCase()} options found. Add one above.</p>
   355→          </CardContent>
   356→        ) : (
   357→          <div className="divide-y">
   358→            {options.map((opt, idx) => (
   359→              <div key={opt.id} className="px-4 py-3 flex items-center gap-3 hover:bg-slate-50 transition-colors">
   360→                <span className="w-8 h-8 rounded-lg bg-slate-100 flex items-center justify-center text-xs font-semibold text-slate-600">{idx + 1}</span>
   361→                {editingId === opt.id ? (
   362→                  <div className="flex-1 flex items-center gap-2">
   363→                    <Input
   364→                      value={editingValue}
   365→                      onChange={(e) => setEditingValue(e.target.value)}
   366→                      onKeyDown={(e) => e.key === "Enter" && handleUpdate()}
   367→                      className="h-8 flex-1"
   368→                      autoFocus
   369→                    />
   370→                    <Button size="sm" onClick={handleUpdate} disabled={loading} className="gap-1 h-8 bg-emerald-600 hover:bg-emerald-500">
   371→                      {loading ? <Loader2 className="w-3.5 h-3.5 animate-spin" /> : <Check className="w-3.5 h-3.5" />} Save
   372→                    </Button>
   373→                    <Button size="sm" variant="ghost" onClick={() => { setEditingId(null); setEditingValue(""); }} className="h-8">Cancel</Button>
   374→                  </div>
   375→                ) : (
   376→                  <>
   377→                    <span className="flex-1 text-sm font-medium text-slate-700">{opt.value}</span>
   378→                    <Button variant="ghost" size="sm" onClick={() => { setEditingId(opt.id); setEditingValue(opt.value); }} className="h-8 w-8 p-0" title="Edit">
   379→                      <Edit3 className="w-3.5 h-3.5 text-blue-600" />
   380→                    </Button>
   381→                    <Button variant="ghost" size="sm" onClick={() => { setDeleteId(opt.id); setDeleteDialogOpen(true); }} className="h-8 w-8 p-0" title="Delete">
   382→                      <Trash2 className="w-3.5 h-3.5 text-red-500" />
   383→                    </Button>
   384→                  </>
   385→                )}
   386→              </div>
   387→            ))}
   388→          </div>
   389→        )}
   390→      </Card>
   391→
   392→      {/* Delete Confirmation Dialog */}
   393→      <Dialog open={deleteDialogOpen} onOpenChange={setDeleteDialogOpen}>
   394→        <DialogContent>
   395→          <DialogHeader>
   396→            <DialogTitle>Delete {label}</DialogTitle>
   397→            <DialogDescription>Are you sure you want to delete this {label.toLowerCase()} option? This action cannot be undone.</DialogDescription>
   398→          </DialogHeader>
   399→          <DialogFooter>
   400→            <Button variant="outline" onClick={() => setDeleteDialogOpen(false)}>Cancel</Button>
   401→            <Button variant="destructive" onClick={handleDelete} disabled={loading}>
   402→              {loading ? <Loader2 className="w-4 h-4 animate-spin mr-2" /> : null} Delete
   403→            </Button>
   404→          </DialogFooter>
   405→        </DialogContent>
   406→      </Dialog>
   407→    </div>
   408→  );
   409→}
