   157→   157→    try {
   158→   158→      const res = await fetch(`/api/dropdown-options?type=${type}`);

   217→   217→      }
   218→   218→    } catch {
   219→   219→      toast({ title: "Error", description: "Failed to delete", variant: "destructive" });
   220→   220→    } finally { setLoading(false); }
   221→   221→  };
   222→   222→
   223→   223→  return (
   224→   224→    <div className="flex items-center gap-1">
   225→   225→      <Select value={value} onValueChange={onChange} disabled={disabled}>
   226→   226→        <SelectTrigger className="h-7 text-xs flex-1">
   227→   227→          <SelectValue placeholder={`Select ${label}...`} />

   231→   231→            <SelectItem key={opt.id} value={opt.value}>{opt.value}</SelectItem>
   232→   232→          ))}
   233→   233→        </SelectContent>
   234→   234→      </Select>
   235→   235→      {!disabled && (
   236→   236→        <Popover open={manageOpen} onOpenChange={setManageOpen}>
   237→   237→          <PopoverTrigger asChild>
   238→   238→            <Button variant="ghost" size="sm" className="h-7 w-7 p-0 shrink-0 text-slate-400 hover:text-slate-700" title="Manage options">
   239→   239→              <Settings2 className="w-3.5 h-3.5" />

   242→   242→          <PopoverContent className="w-72 p-0" align="end">
   243→   243→            <div className="p-3 border-b bg-slate-50">

   245→   245→              <p className="text-xs text-slate-400 mt-0.5">Add, edit or delete options</p>
   246→   246→            </div>
   247→   247→            <div className="p-3 border-b">
   248→   248→              <div className="flex gap-1.5">
   249→   249→                <Input
   250→   250→                  placeholder={`New ${label.toLowerCase()}...`}
   251→   251→                  value={newOptionValue}
   252→   252→                  onChange={(e) => setNewOptionValue(e.target.value)}
   253→   253→                  className="h-7 text-xs"
   254→   254→                  onKeyDown={(e) => e.key === "Enter" && handleAdd()}
   255→   255→                />
   256→   256→                <Button size="sm" onClick={handleAdd} disabled={loading || !newOptionValue.trim()} className="h-7 px-2 shrink-0 bg-slate-700 hover:bg-slate-600">
   257→   257→                  <Plus className="w-3.5 h-3.5" />
   258→   258→                </Button>
   259→   259→              </div>
   260→   260→            </div>
   261→   261→            <div className="max-h-48 overflow-y-auto">
   262→   262→              {options.length === 0 ? (
   263→   263→                <p className="p-3 text-xs text-slate-400 text-center">No options yet</p>
   264→   264→              ) : (
   265→   265→                options.map((opt) => (
   266→   266→                  <div key={opt.id} className="flex items-center gap-1 px-3 py-1.5 border-b last:border-b-0 hover:bg-slate-50 group">
   267→   267→                    {editingId === opt.id ? (
   268→   268→                      <>
   269→   269→                        <Input

   280→   280→                          <X className="w-3 h-3" />

   285→   285→                        <span className="text-xs flex-1 truncate">{opt.value}</span>

   287→   287→                          size="sm" variant="ghost"

   314→   314→// ==================== DatePickerWithAutoClose Component ====================
   315→   315→function DatePickerWithAutoClose({
   316→   316→  value,
   317→   317→  onChange,
   318→   318→}: {
   319→   319→  value: string;
   320→   320→  onChange: (val: string) => void;
   321→   321→}) {
   322→   322→  const [open, setOpen] = useState(false);
   323→   323→
   324→   324→  const parseDate = (dateStr: string): Date | undefined => {
   325→   325→    if (!dateStr) return undefined;
   326→   326→    const parts = dateStr.split("/");
   327→   327→    if (parts.length === 3) {
   328→   328→      return new Date(Number(parts[2]), Number(parts[1]) - 1, Number(parts[0]));
   329→   329→    }
   330→   330→    return undefined;
   331→   331→  };
   332→   332→
   333→   333→  return (
   334→   334→    <Popover open={open} onOpenChange={setOpen}>
   335→   335→      <PopoverTrigger asChild>
   336→   336→        <Button
   337→   337→          variant="outline"
   338→   338→          className="mt-1 w-full justify-start text-left font-normal h-9"
   339→   339→        >
   340→   340→          <CalendarIcon className="mr-2 h-4 w-4 text-slate-500" />
   341→   341→          {value || "Pick a date"}
   342→   342→        </Button>
   343→   343→      </PopoverTrigger>
   344→   344→      <PopoverContent className="w-auto p-0" align="start">
   345→   345→        <Calendar
   346→   346→          mode="single"
   347→   347→          selected={parseDate(value)}
   348→   348→          onSelect={(date) => {
   349→   349→            if (date) {
   350→   350→              onChange(format(date, "dd/MM/yyyy"));
   351→   351→              setOpen(false);
   352→   352→            }
   353→   353→          }}
   354→   354→          captionLayout="dropdown-buttons"
   355→   355→          defaultMonth={parseDate(value) || new Date()}
   356→   356→        />
   357→   357→      </PopoverContent>
   358→   358→    </Popover>
   359→   359→  );
   360→   360→}
