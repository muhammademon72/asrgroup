   129→   129→function DropdownWithManage({
   130→   130→  label,
   131→   131→  type,
   132→   132→  value,
   133→   133→  onChange,
   134→   134→  disabled,
   135→   135→}: {
   136→   136→  label: string;
   137→   137→  type: "department" | "branch" | "address";
   138→   138→  value: string;
   139→   139→  onChange: (val: string) => void;
   140→   140→  disabled: boolean;
   141→   141→}) {
   142→   142→  const [options, setOptions] = useState<DropdownOption[]>([]);
   143→   143→  const [manageOpen, setManageOpen] = useState(false);
   144→   144→  const [newOptionValue, setNewOptionValue] = useState("");
   145→   145→  const [editingId, setEditingId] = useState<string | null>(null);
   146→   146→  const [editingValue, setEditingValue] = useState("");
   147→   147→  const [loading, setLoading] = useState(false);
   148→   148→  const { toast } = useToast();
   149→   149→
   150→   150→  const fetchOptions = useCallback(async () => {
   151→   151→    try {
   152→   152→      const res = await fetch(`/api/dropdown-options?type=${type}`);
   153→   153→      if (res.ok) setOptions(await res.json());
   154→   154→    } catch { /* ignore */ }
   155→   155→  }, [type]);
   156→   156→
   157→   157→  useEffect(() => { fetchOptions(); }, [fetchOptions]);
   158→   158→
   159→   159→  const handleAdd = async () => {
   160→   160→    if (!newOptionValue.trim()) return;
   161→   161→    setLoading(true);
   162→   162→    try {
   163→   163→      const res = await fetch("/api/dropdown-options", {
   164→   164→        method: "POST",
   165→   165→        headers: { "Content-Type": "application/json" },
   166→   166→        body: JSON.stringify({ type, value: newOptionValue.trim() }),
   167→   167→      });
   168→   168→      if (res.ok) {
   169→   169→        toast({ title: "Added", description: `"${newOptionValue.trim()}" added to ${label}` });
   170→   170→        setNewOptionValue("");
   171→   171→        await fetchOptions();
   172→   172→      } else {
   173→   173→        const err = await res.json();
   174→   174→        toast({ title: "Error", description: err.error || "Failed to add", variant: "destructive" });
   175→   175→      }
   176→   176→    } catch {
   177→   177→      toast({ title: "Error", description: "Failed to add option", variant: "destructive" });
   178→   178→    } finally { setLoading(false); }
   179→   179→  };
   180→   180→
   181→   181→  const handleUpdate = async (id: string) => {
   182→   182→    if (!editingValue.trim()) return;
   183→   183→    setLoading(true);
   184→   184→    try {
   185→   185→      const res = await fetch(`/api/dropdown-options/${id}`, {
   186→   186→        method: "PUT",
   187→   187→        headers: { "Content-Type": "application/json" },
   188→   188→        body: JSON.stringify({ value: editingValue.trim() }),
   189→   189→      });
   190→   190→      if (res.ok) {
   191→   191→        toast({ title: "Updated", description: "Option updated successfully" });
   192→   192→        setEditingId(null);
   193→   193→        setEditingValue("");
   194→   194→        await fetchOptions();
   195→   195→      } else {
   196→   196→        toast({ title: "Error", description: "Failed to update", variant: "destructive" });
   197→   197→      }
   198→   198→    } catch {
   199→   199→      toast({ title: "Error", description: "Failed to update", variant: "destructive" });
   200→   200→    } finally { setLoading(false); }
   201→   201→  };
   202→   202→
   203→   203→  const handleDelete = async (id: string, val: string) => {
   204→   204→    setLoading(true);
   205→   205→    try {
   206→   206→      const res = await fetch(`/api/dropdown-options/${id}`, { method: "DELETE" });
   207→   207→      if (res.ok) {
   208→   208→        toast({ title: "Deleted", description: `"${val}" removed from ${label}` });
   209→   209→        if (value === val) onChange("");
   210→   210→        await fetchOptions();
   211→   211→      }
   212→   212→    } catch {
   213→   213→      toast({ title: "Error", description: "Failed to delete", variant: "destructive" });
   214→   214→    } finally { setLoading(false); }
   215→   215→  };
   216→   216→
   217→   217→  return (
   218→   218→    <div className="flex items-center gap-1">
   219→   219→      <Select value={value} onValueChange={onChange} disabled={disabled}>
   220→   220→        <SelectTrigger className="h-7 text-xs flex-1">
   221→   221→          <SelectValue placeholder={`Select ${label}...`} />
   222→   222→        </SelectTrigger>
   223→   223→        <SelectContent>
   224→   224→          {options.map((opt) => (
   225→   225→            <SelectItem key={opt.id} value={opt.value}>{opt.value}</SelectItem>
   226→   226→          ))}
   227→   227→        </SelectContent>
   228→   228→      </Select>
   229→   229→      {!disabled && (
   230→   230→        <Popover open={manageOpen} onOpenChange={setManageOpen}>
   231→   231→          <PopoverTrigger asChild>
   232→   232→            <Button variant="ghost" size="sm" className="h-7 w-7 p-0 shrink-0 text-slate-400 hover:text-slate-700" title="Manage options">
   233→   233→              <Settings2 className="w-3.5 h-3.5" />
   234→   234→            </Button>
   235→   235→          </PopoverTrigger>
   236→   236→          <PopoverContent className="w-72 p-0" align="end">
   237→   237→            <div className="p-3 border-b bg-slate-50">
   238→   238→              <h4 className="text-sm font-semibold text-slate-700">Manage {label}</h4>
   239→   239→              <p className="text-xs text-slate-400 mt-0.5">Add, edit or delete options</p>
   240→   240→            </div>
   241→   241→            <div className="p-3 border-b">
   242→   242→              <div className="flex gap-1.5">
   243→   243→                <Input
   244→   244→                  placeholder={`New ${label.toLowerCase()}...`}
   245→   245→                  value={newOptionValue}
   246→   246→                  onChange={(e) => setNewOptionValue(e.target.value)}
   247→   247→                  className="h-7 text-xs"
   248→   248→                  onKeyDown={(e) => e.key === "Enter" && handleAdd()}
   249→   249→                />
   250→   250→                <Button size="sm" onClick={handleAdd} disabled={loading || !newOptionValue.trim()} className="h-7 px-2 shrink-0 bg-slate-700 hover:bg-slate-600">
   251→   251→                  <Plus className="w-3.5 h-3.5" />
   252→   252→                </Button>
   253→   253→              </div>
   254→   254→            </div>
   255→   255→            <div className="max-h-48 overflow-y-auto">
   256→   256→              {options.length === 0 ? (
   257→   257→                <p className="p-3 text-xs text-slate-400 text-center">No options yet</p>
   258→   258→              ) : (
   259→   259→                options.map((opt) => (
   260→   260→                  <div key={opt.id} className="flex items-center gap-1 px-3 py-1.5 border-b last:border-b-0 hover:bg-slate-50 group">
   261→   261→                    {editingId === opt.id ? (
   262→   262→                      <>
   263→   263→                        <Input
   264→   264→                          value={editingValue}
   265→   265→                          onChange={(e) => setEditingValue(e.target.value)}
   266→   266→                          className="h-6 text-xs flex-1"
   267→   267→                          onKeyDown={(e) => e.key === "Enter" && handleUpdate(opt.id)}
   268→   268→                          autoFocus
   269→   269→                        />
   270→   270→                        <Button size="sm" onClick={() => handleUpdate(opt.id)} disabled={loading} className="h-6 w-6 p-0 shrink-0 text-green-600 hover:text-green-700">
   271→   271→                          <Check className="w-3 h-3" />
   272→   272→                        </Button>
   273→   273→                        <Button size="sm" variant="ghost" onClick={() => { setEditingId(null); setEditingValue(""); }} className="h-6 w-6 p-0 shrink-0">
   274→   274→                          <X className="w-3 h-3" />
   275→   275→                        </Button>
   276→   276→                      </>
   277→   277→                    ) : (
   278→   278→                      <>
   279→   279→                        <span className="text-xs flex-1 truncate">{opt.value}</span>
   280→   280→                        <Button
   281→   281→                          size="sm" variant="ghost"
   282→   282→                          onClick={() => { setEditingId(opt.id); setEditingValue(opt.value); }}
   283→   283→                          className="h-6 w-6 p-0 shrink-0 opacity-0 group-hover:opacity-100 text-blue-500 hover:text-blue-700"
   284→   284→                        >
   285→   285→                          <Pencil className="w-3 h-3" />
   286→   286→                        </Button>
   287→   287→                        <Button
   288→   288→                          size="sm" variant="ghost"
   289→   289→                          onClick={() => handleDelete(opt.id, opt.value)}
   290→   290→                          disabled={loading}
   291→   291→                          className="h-6 w-6 p-0 shrink-0 opacity-0 group-hover:opacity-100 text-red-500 hover:text-red-700"
   292→   292→                        >
   293→   293→                          <Trash2 className="w-3 h-3" />
   294→   294→                        </Button>
   295→   295→                      </>
   296→   296→                    )}
   297→   297→                  </div>
   298→   298→                ))
   299→   299→              )}
   300→   300→            </div>
   301→   301→          </PopoverContent>
   302→   302→        </Popover>
   303→   303→      )}
   304→   304→    </div>
   305→   305→  );
   306→   306→}
