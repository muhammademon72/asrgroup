   143→// ==================== DropdownWithManage Component ====================
   144→function DropdownWithManage({
   145→  label,
   146→  type,
   147→  value,
   148→  onChange,
   149→  disabled,
   150→  isAdmin,
   151→}: {
   152→  label: string;
   153→  type: "department" | "branch" | "address";
   154→  value: string;
   155→  onChange: (val: string) => void;
   156→  disabled: boolean;
   157→  isAdmin: boolean;
   158→}) {
   159→  const [options, setOptions] = useState<DropdownOption[]>([]);
   160→  const [manageOpen, setManageOpen] = useState(false);
   161→  const [newOption, setNewOption] = useState("");
   162→  const [editingOpt, setEditingOpt] = useState<{ id: string; value: string } | null>(null);
   163→  const [deleteOptId, setDeleteOptId] = useState<string | null>(null);
   164→  const [deleteDialogOpen, setDeleteDialogOpen] = useState(false);
   165→  const [manageLoading, setManageLoading] = useState(false);
   166→  const { toast } = useToast();
   167→
   168→  const fetchOptions = useCallback(async () => {
   169→    try {
   170→      const res = await fetch(`/api/dropdown-options?type=${type}`);
   171→      if (res.ok) setOptions(await res.json());
   172→    } catch { /* ignore */ }
   173→  }, [type]);
   174→
   175→  useEffect(() => { fetchOptions(); }, [fetchOptions]);
   176→
   177→  const handleAddOption = async () => {
   178→    if (!newOption.trim()) return;
   179→    setManageLoading(true);
   180→    try {
   181→      const res = await fetch("/api/dropdown-options", {
   182→        method: "POST",
   183→        headers: { "Content-Type": "application/json" },
   184→        body: JSON.stringify({ type, value: newOption.trim() }),
   185→      });
   186→      if (res.ok) {
   187→        toast({ title: "Added", description: `"${newOption.trim()}" added successfully` });
   188→        setNewOption("");
   189→        await fetchOptions();
   190→      } else {
   191→        const err = await res.json();
   192→        toast({ title: "Error", description: err.error || "Failed to add", variant: "destructive" });
   193→      }
   194→    } catch {
   195→      toast({ title: "Error", description: "Failed to add option", variant: "destructive" });
   196→    } finally { setManageLoading(false); }
   197→  };
   198→
   199→  const handleUpdateOption = async () => {
   200→    if (!editingOpt || !editingOpt.value.trim()) return;
   201→    setManageLoading(true);
   202→    try {
   203→      const res = await fetch(`/api/dropdown-options/${editingOpt.id}`, {
   204→        method: "PUT",
   205→        headers: { "Content-Type": "application/json" },
   206→        body: JSON.stringify({ value: editingOpt.value.trim() }),
   207→      });
   208→      if (res.ok) {
   209→        toast({ title: "Updated", description: "Option updated successfully" });
   210→        setEditingOpt(null);
   211→        await fetchOptions();
   212→      } else {
   213→        toast({ title: "Error", description: "Failed to update", variant: "destructive" });
   214→      }
   215→    } catch {
   216→      toast({ title: "Error", description: "Failed to update", variant: "destructive" });
   217→    } finally { setManageLoading(false); }
   218→  };
   219→
   220→  const handleDeleteOption = async () => {
   221→    if (!deleteOptId) return;
   222→    setManageLoading(true);
   223→    try {
   224→      const res = await fetch(`/api/dropdown-options/${deleteOptId}`, { method: "DELETE" });
   225→      if (res.ok) {
   226→        toast({ title: "Deleted", description: "Option removed successfully" });
   227→        setDeleteOptId(null);
   228→        setDeleteDialogOpen(false);
   229→        await fetchOptions();
   230→      }
   231→    } catch {
   232→      toast({ title: "Error", description: "Failed to delete", variant: "destructive" });
   233→    } finally { setManageLoading(false); }
   234→  };
   235→
   236→  return (
   237→    <div className="flex items-center gap-1">
   238→      <Select value={value} onValueChange={onChange} disabled={disabled}>
   239→        <SelectTrigger className="h-7 text-xs flex-1 min-w-0">
   240→          <SelectValue placeholder={`Select ${label}...`} />
   241→        </SelectTrigger>
   242→        <SelectContent>
   243→          {/* Show current value as fallback if it doesn't match any DB option */}
   244→          {value && !options.some(opt => opt.value === value) && (
   245→            <SelectItem key="__current__" value={value}>{value}</SelectItem>
   246→          )}
   247→          {options.map((opt) => (
   248→            <SelectItem key={opt.id} value={opt.value}>{opt.value}</SelectItem>
   249→          ))}
   250→        </SelectContent>
   251→      </Select>
   252→      {isAdmin && (
   253→        <Popover open={manageOpen} onOpenChange={setManageOpen}>
   254→          <PopoverTrigger asChild>
   255→            <Button
   256→              variant="ghost"
   257→              size="sm"
   258→              className="h-7 w-7 p-0 shrink-0 text-slate-400 hover:text-slate-700 hover:bg-slate-100"
   259→              title="Manage options"
   260→            >
   261→              <Settings2 className="w-3.5 h-3.5" />
   262→            </Button>
   263→          </PopoverTrigger>
   264→          <PopoverContent className="w-72 p-0" align="end" side="bottom">
   265→            <div className="p-3 border-b bg-slate-50">
   266→              <h4 className="text-sm font-semibold text-slate-800">Manage {label}</h4>
   267→              <p className="text-xs text-slate-500">Add, edit, or remove options</p>
   268→            </div>
   269→            <div className="p-3 border-b">
   270→              <div className="flex items-center gap-2">
   271→                <Input
   272→                  placeholder={`New ${label.toLowerCase()}...`}
   273→                  value={newOption}
   274→                  onChange={(e) => setNewOption(e.target.value)}
   275→                  onKeyDown={(e) => e.key === "Enter" && handleAddOption()}
   276→                  className="h-8 text-xs flex-1"
   277→                />
   278→                <Button
   279→                  onClick={handleAddOption}
   280→                  disabled={manageLoading || !newOption.trim()}
   281→                  size="sm"
   282→                  className="h-8 gap-1 bg-slate-800 hover:bg-slate-700 shrink-0"
   283→                >
   284→                  {manageLoading ? <Loader2 className="w-3 h-3 animate-spin" /> : <Plus className="w-3 h-3" />}
   285→                  Add
   286→                </Button>
   287→              </div>
   288→            </div>
   289→            <div className="max-h-48 overflow-y-auto">
   290→              {options.length === 0 ? (
   291→                <div className="p-4 text-center text-xs text-slate-400">No options yet</div>
   292→              ) : (
   293→                options.map((opt) => (
   294→                  <div key={opt.id} className="flex items-center gap-2 px-3 py-2 border-b last:border-b-0 hover:bg-slate-50 transition-colors">
   295→                    {editingOpt?.id === opt.id ? (
   296→                      <>
   297→                        <Input
   298→                          value={editingOpt.value}
   299→                          onChange={(e) => setEditingOpt({ ...editingOpt, value: e.target.value })}
   300→                          onKeyDown={(e) => e.key === "Enter" && handleUpdateOption()}
   301→                          className="h-7 text-xs flex-1"
   302→                          autoFocus
   303→                        />
   304→                        <Button size="sm" onClick={handleUpdateOption} disabled={manageLoading} className="h-7 w-7 p-0 bg-emerald-600 hover:bg-emerald-500 shrink-0">
   305→                          <Check className="w-3 h-3" />
   306→                        </Button>
   307→                        <Button size="sm" variant="ghost" onClick={() => setEditingOpt(null)} className="h-7 w-7 p-0 shrink-0">
   308→                          <X className="w-3 h-3" />
   309→                        </Button>
   310→                      </>
   311→                    ) : (
   312→                      <>
   313→                        <span className="flex-1 text-xs text-slate-700 truncate">{opt.value}</span>
   314→                        <Button variant="ghost" size="sm" onClick={() => setEditingOpt({ id: opt.id, value: opt.value })} className="h-7 w-7 p-0 shrink-0" title="Edit">
   315→                          <Pencil className="w-3 h-3 text-blue-500" />
   316→                        </Button>
   317→                        <Button variant="ghost" size="sm" onClick={() => { setDeleteOptId(opt.id); setDeleteDialogOpen(true); }} className="h-7 w-7 p-0 shrink-0" title="Delete">
   318→                          <Trash2 className="w-3 h-3 text-red-400" />
   319→                        </Button>
   320→                      </>
   321→                    )}
   322→                  </div>
   323→                ))
   324→              )}
   325→            </div>
   326→          </PopoverContent>
   327→        </Popover>
   328→      )}
   329→      {/* Delete Confirmation Dialog */}
   330→      <Dialog open={deleteDialogOpen} onOpenChange={setDeleteDialogOpen}>
   331→        <DialogContent>
   332→          <DialogHeader>
   333→            <DialogTitle>Delete {label} Option</DialogTitle>
   334→            <DialogDescription>Are you sure you want to delete this option? This action cannot be undone.</DialogDescription>
   335→          </DialogHeader>
   336→          <DialogFooter>
   337→            <Button variant="outline" onClick={() => setDeleteDialogOpen(false)}>Cancel</Button>
   338→            <Button variant="destructive" onClick={handleDeleteOption} disabled={manageLoading}>
   339→              {manageLoading ? <Loader2 className="w-4 h-4 animate-spin mr-2" /> : null} Delete
   340→            </Button>
   341→          </DialogFooter>
   342→        </DialogContent>
   343→      </Dialog>
   344→    </div>
   345→  );
   346→}

   348→// ==================== DatePickerWithAutoClose Component ====================
   349→function DatePickerWithAutoClose({
   350→  value,
   351→  onChange,
   352→}: {
   353→  value: string;
   354→  onChange: (val: string) => void;
   355→}) {
   356→  const [open, setOpen] = useState(false);
   357→
   358→  const parseDate = (dateStr: string): Date | undefined => {
   359→    if (!dateStr) return undefined;
   360→    const parts = dateStr.split("/");
   361→    if (parts.length === 3) {
   362→      return new Date(Number(parts[2]), Number(parts[1]) - 1, Number(parts[0]));
   363→    }
   364→    return undefined;
   365→  };
   366→
   367→  return (
   368→    <Popover open={open} onOpenChange={setOpen}>
   369→      <PopoverTrigger asChild>
   370→        <Button
   371→          variant="outline"
   372→          className="mt-1 w-full justify-start text-left font-normal h-9"
   373→        >
   374→          <CalendarIcon className="mr-2 h-4 w-4 text-slate-500" />
   375→          {value || "Pick a date"}
   376→        </Button>
   377→      </PopoverTrigger>
   378→      <PopoverContent className="w-auto p-0" align="start">
   379→        <Calendar
   380→          mode="single"
   381→          selected={parseDate(value)}
   382→          onSelect={(date) => {
   383→            if (date) {
   384→              onChange(format(date, "dd/MM/yyyy"));
   385→              setOpen(false);
   386→            }
   387→          }}
   388→          captionLayout="dropdown-buttons"
   389→          defaultMonth={parseDate(value) || new Date()}
   390→        />
   391→      </PopoverContent>
   392→    </Popover>
   393→  );
   394→}
