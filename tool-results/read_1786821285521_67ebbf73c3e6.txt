   138→// ==================== DropdownWithManage Component ====================
   139→function DropdownWithManage({
   140→  label,
   141→  type,
   142→  value,
   143→  onChange,
   144→  disabled,
   145→  isAdmin,
   146→}: {
   147→  label: string;
   148→  type: "department" | "branch" | "address";
   149→  value: string;
   150→  onChange: (val: string) => void;
   151→  disabled: boolean;
   152→  isAdmin: boolean;
   153→}) {
   154→  const [options, setOptions] = useState<DropdownOption[]>([]);
   155→  const [manageOpen, setManageOpen] = useState(false);
   156→  const [newOptionValue, setNewOptionValue] = useState("");
   157→  const [editingId, setEditingId] = useState<string | null>(null);
   158→  const [editingValue, setEditingValue] = useState("");
   159→  const [loading, setLoading] = useState(false);
   160→  const { toast } = useToast();
   161→
   162→  const fetchOptions = useCallback(async () => {
   163→    try {
   164→      const res = await fetch(`/api/dropdown-options?type=${type}`);
   165→      if (res.ok) setOptions(await res.json());
   166→    } catch { /* ignore */ }
   167→  }, [type]);
   168→
   169→  useEffect(() => { fetchOptions(); }, [fetchOptions]);
   170→
   171→  const handleAdd = async () => {
   172→    if (!newOptionValue.trim()) return;
   173→    setLoading(true);
   174→    try {
   175→      const res = await fetch("/api/dropdown-options", {
   176→        method: "POST",
   177→        headers: { "Content-Type": "application/json" },
   178→        body: JSON.stringify({ type, value: newOptionValue.trim() }),
   179→      });
   180→      if (res.ok) {
   181→        toast({ title: "Added", description: `"${newOptionValue.trim()}" added to ${label}` });
   182→        setNewOptionValue("");
   183→        await fetchOptions();
   184→      } else {
   185→        const err = await res.json();
   186→        toast({ title: "Error", description: err.error || "Failed to add", variant: "destructive" });
   187→      }
   188→    } catch {
   189→      toast({ title: "Error", description: "Failed to add option", variant: "destructive" });
   190→    } finally { setLoading(false); }
   191→  };
   192→
   193→  const handleUpdate = async (id: string) => {
   194→    if (!editingValue.trim()) return;
   195→    setLoading(true);
   196→    try {
   197→      const res = await fetch(`/api/dropdown-options/${id}`, {
   198→        method: "PUT",
   199→        headers: { "Content-Type": "application/json" },
   200→        body: JSON.stringify({ value: editingValue.trim() }),
   201→      });
   202→      if (res.ok) {
   203→        toast({ title: "Updated", description: "Option updated successfully" });
   204→        setEditingId(null);
   205→        setEditingValue("");
   206→        await fetchOptions();
   207→      } else {
   208→        toast({ title: "Error", description: "Failed to update", variant: "destructive" });
   209→      }
   210→    } catch {
   211→      toast({ title: "Error", description: "Failed to update", variant: "destructive" });
   212→    } finally { setLoading(false); }
   213→  };
   214→
   215→  const handleDelete = async (id: string, val: string) => {
   216→    setLoading(true);
   217→    try {
   218→      const res = await fetch(`/api/dropdown-options/${id}`, { method: "DELETE" });
   219→      if (res.ok) {
   220→        toast({ title: "Deleted", description: `"${val}" removed from ${label}` });
   221→        if (value === val) onChange("");
   222→        await fetchOptions();
   223→      }
   224→    } catch {
   225→      toast({ title: "Error", description: "Failed to delete", variant: "destructive" });
   226→    } finally { setLoading(false); }
   227→  };
   228→
   229→  return (
   230→    <div className="flex items-center gap-1">
   231→      <Select value={value} onValueChange={onChange} disabled={disabled}>
   232→        <SelectTrigger className="h-7 text-xs flex-1">
   233→          <SelectValue placeholder={`Select ${label}...`} />
   234→        </SelectTrigger>
   235→        <SelectContent>
   236→          {options.map((opt) => (
   237→            <SelectItem key={opt.id} value={opt.value}>{opt.value}</SelectItem>
   238→          ))}
   239→        </SelectContent>
   240→      </Select>
   241→      {!disabled && isAdmin && (
   242→        <Popover open={manageOpen} onOpenChange={setManageOpen}>
   243→          <PopoverTrigger asChild>
   244→            <Button variant="ghost" size="sm" className="h-7 w-7 p-0 shrink-0 text-slate-400 hover:text-slate-700" title="Manage options">
   245→              <Settings2 className="w-3.5 h-3.5" />
   246→            </Button>
   247→          </PopoverTrigger>
   248→          <PopoverContent className="w-72 p-0" align="end">
   249→            <div className="p-3 border-b bg-slate-50">
   250→              <h4 className="text-sm font-semibold text-slate-700">Manage {label}</h4>
   251→              <p className="text-xs text-slate-400 mt-0.5">Add, edit or delete options</p>
   252→            </div>
   253→            <div className="p-3 border-b">
   254→              <div className="flex gap-1.5">
   255→                <Input
   256→                  placeholder={`New ${label.toLowerCase()}...`}
   257→                  value={newOptionValue}
   258→                  onChange={(e) => setNewOptionValue(e.target.value)}
   259→                  className="h-7 text-xs"
   260→                  onKeyDown={(e) => e.key === "Enter" && handleAdd()}
   261→                />
   262→                <Button size="sm" onClick={handleAdd} disabled={loading || !newOptionValue.trim()} className="h-7 px-2 shrink-0 bg-slate-700 hover:bg-slate-600">
   263→                  <Plus className="w-3.5 h-3.5" />
   264→                </Button>
   265→              </div>
   266→            </div>
   267→            <div className="max-h-48 overflow-y-auto">
   268→              {options.length === 0 ? (
   269→                <p className="p-3 text-xs text-slate-400 text-center">No options yet</p>
   270→              ) : (
   271→                options.map((opt) => (
   272→                  <div key={opt.id} className="flex items-center gap-1 px-3 py-1.5 border-b last:border-b-0 hover:bg-slate-50 group">
   273→                    {editingId === opt.id ? (
   274→                      <>
   275→                        <Input
   276→                          value={editingValue}
   277→                          onChange={(e) => setEditingValue(e.target.value)}
   278→                          className="h-6 text-xs flex-1"
   279→                          onKeyDown={(e) => e.key === "Enter" && handleUpdate(opt.id)}
   280→                          autoFocus
   281→                        />
   282→                        <Button size="sm" onClick={() => handleUpdate(opt.id)} disabled={loading} className="h-6 w-6 p-0 shrink-0 text-green-600 hover:text-green-700">
   283→                          <Check className="w-3 h-3" />
   284→                        </Button>
   285→                        <Button size="sm" variant="ghost" onClick={() => { setEditingId(null); setEditingValue(""); }} className="h-6 w-6 p-0 shrink-0">
   286→                          <X className="w-3 h-3" />
   287→                        </Button>
   288→                      </>
   289→                    ) : (
   290→                      <>
   291→                        <span className="text-xs flex-1 truncate">{opt.value}</span>
   292→                        <Button
   293→                          size="sm" variant="ghost"
   294→                          onClick={() => { setEditingId(opt.id); setEditingValue(opt.value); }}
   295→                          className="h-6 w-6 p-0 shrink-0 opacity-0 group-hover:opacity-100 text-blue-500 hover:text-blue-700"
   296→                        >
   297→                          <Pencil className="w-3 h-3" />
   298→                        </Button>
   299→                        <Button
   300→                          size="sm" variant="ghost"
   301→                          onClick={() => handleDelete(opt.id, opt.value)}
   302→                          disabled={loading}
   303→                          className="h-6 w-6 p-0 shrink-0 opacity-0 group-hover:opacity-100 text-red-500 hover:text-red-700"
   304→                        >
   305→                          <Trash2 className="w-3 h-3" />
   306→                        </Button>
   307→                      </>
   308→                    )}
   309→                  </div>
   310→                ))
   311→              )}
   312→            </div>
   313→          </PopoverContent>
   314→        </Popover>
   315→      )}
   316→    </div>
   317→  );
   318→}

   320→// ==================== DatePickerWithAutoClose Component ====================
   321→function DatePickerWithAutoClose({
   322→  value,
   323→  onChange,
   324→}: {
   325→  value: string;
   326→  onChange: (val: string) => void;
   327→}) {
   328→  const [open, setOpen] = useState(false);
   329→
   330→  const parseDate = (dateStr: string): Date | undefined => {
   331→    if (!dateStr) return undefined;
   332→    const parts = dateStr.split("/");
   333→    if (parts.length === 3) {
   334→      return new Date(Number(parts[2]), Number(parts[1]) - 1, Number(parts[0]));
   335→    }
   336→    return undefined;
   337→  };
   338→
   339→  return (
   340→    <Popover open={open} onOpenChange={setOpen}>
   341→      <PopoverTrigger asChild>
   342→        <Button
   343→          variant="outline"
   344→          className="mt-1 w-full justify-start text-left font-normal h-9"
   345→        >
   346→          <CalendarIcon className="mr-2 h-4 w-4 text-slate-500" />
   347→          {value || "Pick a date"}
   348→        </Button>
   349→      </PopoverTrigger>
   350→      <PopoverContent className="w-auto p-0" align="start">
   351→        <Calendar
   352→          mode="single"
   353→          selected={parseDate(value)}
   354→          onSelect={(date) => {
   355→            if (date) {
   356→              onChange(format(date, "dd/MM/yyyy"));
   357→              setOpen(false);
   358→            }
   359→          }}
   360→          captionLayout="dropdown-buttons"
   361→          defaultMonth={parseDate(value) || new Date()}
   362→        />
   363→      </PopoverContent>
   364→    </Popover>
   365→  );
   366→}
