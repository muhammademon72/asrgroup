   148→   148→  type: "department" | "branch" | "address";
   149→   149→  value: string;
   150→   150→  onChange: (val: string) => void;

   152→   152→  isAdmin: boolean;

   154→   154→  const [options, setOptions] = useState<DropdownOption[]>([]);
   155→   155→  const [manageOpen, setManageOpen] = useState(false);

   162→   162→  const fetchOptions = useCallback(async () => {
   163→   163→    try {
   164→   164→      const res = await fetch(`/api/dropdown-options?type=${type}`);
   165→   165→      if (res.ok) setOptions(await res.json());
   166→   166→    } catch { /* ignore */ }

   168→   168→
   169→   169→  useEffect(() => { fetchOptions(); }, [fetchOptions]);

   172→   172→    if (!newOptionValue.trim()) return;

   175→   175→      const res = await fetch("/api/dropdown-options", {
   176→   176→        method: "POST",
   177→   177→        headers: { "Content-Type": "application/json" },
   178→   178→        body: JSON.stringify({ type, value: newOptionValue.trim() }),
   179→   179→      });
   180→   180→      if (res.ok) {
   181→   181→        toast({ title: "Added", description: `"${newOptionValue.trim()}" added to ${label}` });
   182→   182→        setNewOptionValue("");
   183→   183→        await fetchOptions();
   184→   184→      } else {
   185→   185→        const err = await res.json();
   186→   186→        toast({ title: "Error", description: err.error || "Failed to add", variant: "destructive" });
   187→   187→      }

   189→   189→      toast({ title: "Error", description: "Failed to add option", variant: "destructive" });

   197→   197→      const res = await fetch(`/api/dropdown-options/${id}`, {
   198→   198→        method: "PUT",
   199→   199→        headers: { "Content-Type": "application/json" },
   200→   200→        body: JSON.stringify({ value: editingValue.trim() }),
   201→   201→      });

   203→   203→        toast({ title: "Updated", description: "Option updated successfully" });
   204→   204→        setEditingId(null);
   205→   205→        setEditingValue("");

   207→   207→      } else {
   208→   208→        toast({ title: "Error", description: "Failed to update", variant: "destructive" });
   209→   209→      }

   218→   218→      const res = await fetch(`/api/dropdown-options/${id}`, { method: "DELETE" });

   220→   220→        toast({ title: "Deleted", description: `"${val}" removed from ${label}` });
   221→   221→        if (value === val) onChange("");
   222→   222→        await fetchOptions();
   223→   223→      }
   224→   224→    } catch {
   225→   225→      toast({ title: "Error", description: "Failed to delete", variant: "destructive" });
   226→   226→    } finally { setLoading(false); }
   227→   227→  };
   228→   228→
   229→   229→  return (
   230→   230→    <div className="flex items-center gap-1">
   231→   231→      <Select value={value} onValueChange={onChange} disabled={disabled}>
   232→   232→        <SelectTrigger className="h-7 text-xs flex-1">
   233→   233→          <SelectValue placeholder={`Select ${label}...`} />
   234→   234→        </SelectTrigger>
   235→   235→        <SelectContent>
   236→   236→          {options.map((opt) => (
   237→   237→            <SelectItem key={opt.id} value={opt.value}>{opt.value}</SelectItem>
   238→   238→          ))}
   239→   239→        </SelectContent>
   240→   240→      </Select>
   241→   241→      {!disabled && isAdmin && (
   242→   242→        <Popover open={manageOpen} onOpenChange={setManageOpen}>
   243→   243→          <PopoverTrigger asChild>
   244→   244→            <Button variant="ghost" size="sm" className="h-7 w-7 p-0 shrink-0 text-slate-400 hover:text-slate-700" title="Manage options">
   245→   245→              <Settings2 className="w-3.5 h-3.5" />
   246→   246→            </Button>
   247→   247→          </PopoverTrigger>
   248→   248→          <PopoverContent className="w-72 p-0" align="end">
   249→   249→            <div className="p-3 border-b bg-slate-50">
   250→   250→              <h4 className="text-sm font-semibold text-slate-700">Manage {label}</h4>
   251→   251→              <p className="text-xs text-slate-400 mt-0.5">Add, edit or delete options</p>
   252→   252→            </div>
   253→   253→            <div className="p-3 border-b">
   254→   254→              <div className="flex gap-1.5">
   255→   255→                <Input
   256→   256→                  placeholder={`New ${label.toLowerCase()}...`}
   257→   257→                  value={newOptionValue}
   258→   258→                  onChange={(e) => setNewOptionValue(e.target.value)}
   259→   259→                  className="h-7 text-xs"
   260→   260→                  onKeyDown={(e) => e.key === "Enter" && handleAdd()}
   261→   261→                />
   262→   262→                <Button size="sm" onClick={handleAdd} disabled={loading || !newOptionValue.trim()} className="h-7 px-2 shrink-0 bg-slate-700 hover:bg-slate-600">
   263→   263→                  <Plus className="w-3.5 h-3.5" />
   264→   264→                </Button>
   265→   265→              </div>
   266→   266→            </div>
   267→   267→            <div className="max-h-48 overflow-y-auto">

   269→   269→                <p className="p-3 text-xs text-slate-400 text-center">No options yet</p>
   270→   270→              ) : (
   271→   271→                options.map((opt) => (
   272→   272→                  <div key={opt.id} className="flex items-center gap-1 px-3 py-1.5 border-b last:border-b-0 hover:bg-slate-50 group">
   273→   273→                    {editingId === opt.id ? (
   274→   274→                      <>
   275→   275→                        <Input
   276→   276→                          value={editingValue}

   283→   283→                          <Check className="w-3 h-3" />
   284→   284→                        </Button>

   291→   291→                        <span className="text-xs flex-1 truncate">{opt.value}</span>
   292→   292→                        <Button
   293→   293→                          size="sm" variant="ghost"
   294→   294→                          onClick={() => { setEditingId(opt.id); setEditingValue(opt.value); }}

   320→   320→// ==================== DatePickerWithAutoClose Component ====================
   321→   321→function DatePickerWithAutoClose({
   322→   322→  value,
   323→   323→  onChange,
   324→   324→}: {
   325→   325→  value: string;
   326→   326→  onChange: (val: string) => void;
   327→   327→}) {
   328→   328→  const [open, setOpen] = useState(false);
   329→   329→
   330→   330→  const parseDate = (dateStr: string): Date | undefined => {
   331→   331→    if (!dateStr) return undefined;
   332→   332→    const parts = dateStr.split("/");
   333→   333→    if (parts.length === 3) {
   334→   334→      return new Date(Number(parts[2]), Number(parts[1]) - 1, Number(parts[0]));
   335→   335→    }
   336→   336→    return undefined;
   337→   337→  };
   338→   338→
   339→   339→  return (
   340→   340→    <Popover open={open} onOpenChange={setOpen}>
   341→   341→      <PopoverTrigger asChild>
   342→   342→        <Button
   343→   343→          variant="outline"
   344→   344→          className="mt-1 w-full justify-start text-left font-normal h-9"
   345→   345→        >
   346→   346→          <CalendarIcon className="mr-2 h-4 w-4 text-slate-500" />
   347→   347→          {value || "Pick a date"}
   348→   348→        </Button>
   349→   349→      </PopoverTrigger>
   350→   350→      <PopoverContent className="w-auto p-0" align="start">
   351→   351→        <Calendar
   352→   352→          mode="single"
   353→   353→          selected={parseDate(value)}
   354→   354→          onSelect={(date) => {
   355→   355→            if (date) {
   356→   356→              onChange(format(date, "dd/MM/yyyy"));
   357→   357→              setOpen(false);
   358→   358→            }
   359→   359→          }}
   360→   360→          captionLayout="dropdown-buttons"
   361→   361→          defaultMonth={parseDate(value) || new Date()}
   362→   362→        />
   363→   363→      </PopoverContent>
   364→   364→    </Popover>
   365→   365→  );
   366→   366→}
