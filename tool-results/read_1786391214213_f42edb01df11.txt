   128→// ==================== DropdownWithManage Component ====================
   129→function DropdownWithManage({
   130→  label,
   131→  type,
   132→  value,
   133→  onChange,
   134→  disabled,
   135→}: {
   136→  label: string;
   137→  type: "department" | "branch" | "address";
   138→  value: string;
   139→  onChange: (val: string) => void;
   140→  disabled: boolean;
   141→}) {
   142→  const [options, setOptions] = useState<DropdownOption[]>([]);
   143→  const [manageOpen, setManageOpen] = useState(false);
   144→  const [newOptionValue, setNewOptionValue] = useState("");
   145→  const [editingId, setEditingId] = useState<string | null>(null);
   146→  const [editingValue, setEditingValue] = useState("");
   147→  const [loading, setLoading] = useState(false);
   148→  const { toast } = useToast();
   149→
   150→  const fetchOptions = useCallback(async () => {
   151→    try {
   152→      const res = await fetch(`/api/dropdown-options?type=${type}`);
   153→      if (res.ok) setOptions(await res.json());
   154→    } catch { /* ignore */ }
   155→  }, [type]);
   156→
   157→  useEffect(() => { fetchOptions(); }, [fetchOptions]);
   158→
   159→  const handleAdd = async () => {
   160→    if (!newOptionValue.trim()) return;
   161→    setLoading(true);
   162→    try {
   163→      const res = await fetch("/api/dropdown-options", {
   164→        method: "POST",
   165→        headers: { "Content-Type": "application/json" },
   166→        body: JSON.stringify({ type, value: newOptionValue.trim() }),
   167→      });
   168→      if (res.ok) {
   169→        toast({ title: "Added", description: `"${newOptionValue.trim()}" added to ${label}` });
   170→        setNewOptionValue("");
   171→        await fetchOptions();
   172→      } else {
   173→        const err = await res.json();
   174→        toast({ title: "Error", description: err.error || "Failed to add", variant: "destructive" });
   175→      }
   176→    } catch {
   177→      toast({ title: "Error", description: "Failed to add option", variant: "destructive" });
   178→    } finally { setLoading(false); }
   179→  };
   180→
   181→  const handleUpdate = async (id: string) => {
   182→    if (!editingValue.trim()) return;
   183→    setLoading(true);
   184→    try {
   185→      const res = await fetch(`/api/dropdown-options/${id}`, {
   186→        method: "PUT",
   187→        headers: { "Content-Type": "application/json" },
   188→        body: JSON.stringify({ value: editingValue.trim() }),
   189→      });
   190→      if (res.ok) {
   191→        toast({ title: "Updated", description: "Option updated successfully" });
   192→        setEditingId(null);
   193→        setEditingValue("");
   194→        await fetchOptions();
   195→      } else {
   196→        toast({ title: "Error", description: "Failed to update", variant: "destructive" });
   197→      }
   198→    } catch {
   199→      toast({ title: "Error", description: "Failed to update", variant: "destructive" });
   200→    } finally { setLoading(false); }
   201→  };
   202→
   203→  const handleDelete = async (id: string, val: string) => {
   204→    setLoading(true);
   205→    try {
   206→      const res = await fetch(`/api/dropdown-options/${id}`, { method: "DELETE" });
   207→      if (res.ok) {
   208→        toast({ title: "Deleted", description: `"${val}" removed from ${label}` });
   209→        if (value === val) onChange("");
   210→        await fetchOptions();
   211→      }
   212→    } catch {
   213→      toast({ title: "Error", description: "Failed to delete", variant: "destructive" });
   214→    } finally { setLoading(false); }
   215→  };
   216→
   217→  return (
   218→    <div className="flex items-center gap-1">
   219→      <Select value={value} onValueChange={onChange} disabled={disabled}>
   220→        <SelectTrigger className="h-7 text-xs flex-1">
   221→          <SelectValue placeholder={`Select ${label}...`} />
   222→        </SelectTrigger>
   223→        <SelectContent>
   224→          {options.map((opt) => (
   225→            <SelectItem key={opt.id} value={opt.value}>{opt.value}</SelectItem>
   226→          ))}
   227→        </SelectContent>
   228→      </Select>
   229→      {!disabled && (
   230→        <Popover open={manageOpen} onOpenChange={setManageOpen}>
   231→          <PopoverTrigger asChild>
   232→            <Button variant="ghost" size="sm" className="h-7 w-7 p-0 shrink-0 text-slate-400 hover:text-slate-700" title="Manage options">
   233→              <Settings2 className="w-3.5 h-3.5" />
   234→            </Button>
   235→          </PopoverTrigger>
   236→          <PopoverContent className="w-72 p-0" align="end">
   237→            <div className="p-3 border-b bg-slate-50">
   238→              <h4 className="text-sm font-semibold text-slate-700">Manage {label}</h4>
   239→              <p className="text-xs text-slate-400 mt-0.5">Add, edit or delete options</p>
   240→            </div>
   241→            <div className="p-3 border-b">
   242→              <div className="flex gap-1.5">
   243→                <Input
   244→                  placeholder={`New ${label.toLowerCase()}...`}
   245→                  value={newOptionValue}
   246→                  onChange={(e) => setNewOptionValue(e.target.value)}
   247→                  className="h-7 text-xs"
   248→                  onKeyDown={(e) => e.key === "Enter" && handleAdd()}
   249→                />
   250→                <Button size="sm" onClick={handleAdd} disabled={loading || !newOptionValue.trim()} className="h-7 px-2 shrink-0 bg-slate-700 hover:bg-slate-600">
   251→                  <Plus className="w-3.5 h-3.5" />
   252→                </Button>
   253→              </div>
   254→            </div>
   255→            <div className="max-h-48 overflow-y-auto">
   256→              {options.length === 0 ? (
   257→                <p className="p-3 text-xs text-slate-400 text-center">No options yet</p>
   258→              ) : (
   259→                options.map((opt) => (
   260→                  <div key={opt.id} className="flex items-center gap-1 px-3 py-1.5 border-b last:border-b-0 hover:bg-slate-50 group">
   261→                    {editingId === opt.id ? (
   262→                      <>
   263→                        <Input
   264→                          value={editingValue}
   265→                          onChange={(e) => setEditingValue(e.target.value)}
   266→                          className="h-6 text-xs flex-1"
   267→                          onKeyDown={(e) => e.key === "Enter" && handleUpdate(opt.id)}
   268→                          autoFocus
   269→                        />
   270→                        <Button size="sm" onClick={() => handleUpdate(opt.id)} disabled={loading} className="h-6 w-6 p-0 shrink-0 text-green-600 hover:text-green-700">
   271→                          <Check className="w-3 h-3" />
   272→                        </Button>
   273→                        <Button size="sm" variant="ghost" onClick={() => { setEditingId(null); setEditingValue(""); }} className="h-6 w-6 p-0 shrink-0">
   274→                          <X className="w-3 h-3" />
   275→                        </Button>
   276→                      </>
   277→                    ) : (
   278→                      <>
   279→                        <span className="text-xs flex-1 truncate">{opt.value}</span>
   280→                        <Button
   281→                          size="sm" variant="ghost"
   282→                          onClick={() => { setEditingId(opt.id); setEditingValue(opt.value); }}
   283→                          className="h-6 w-6 p-0 shrink-0 opacity-0 group-hover:opacity-100 text-blue-500 hover:text-blue-700"
   284→                        >
   285→                          <Pencil className="w-3 h-3" />
   286→                        </Button>
   287→                        <Button
   288→                          size="sm" variant="ghost"
   289→                          onClick={() => handleDelete(opt.id, opt.value)}
   290→                          disabled={loading}
   291→                          className="h-6 w-6 p-0 shrink-0 opacity-0 group-hover:opacity-100 text-red-500 hover:text-red-700"
   292→                        >
   293→                          <Trash2 className="w-3 h-3" />
   294→                        </Button>
   295→                      </>
   296→                    )}
   297→                  </div>
   298→                ))
   299→              )}
   300→            </div>
   301→          </PopoverContent>
   302→        </Popover>
   303→      )}
   304→    </div>
   305→  );
   306→}
