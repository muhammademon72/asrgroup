   131→// ==================== DropdownWithManage Component ====================
   132→function DropdownWithManage({
   133→  label,
   134→  type,
   135→  value,
   136→  onChange,
   137→  disabled,
   138→}: {
   139→  label: string;
   140→  type: "department" | "branch" | "address";
   141→  value: string;
   142→  onChange: (val: string) => void;
   143→  disabled: boolean;
   144→}) {
   145→  const [options, setOptions] = useState<DropdownOption[]>([]);
   146→  const [manageOpen, setManageOpen] = useState(false);
   147→  const [newOptionValue, setNewOptionValue] = useState("");
   148→  const [editingId, setEditingId] = useState<string | null>(null);
   149→  const [editingValue, setEditingValue] = useState("");
   150→  const [loading, setLoading] = useState(false);
   151→  const { toast } = useToast();
   152→
   153→  const fetchOptions = useCallback(async () => {
   154→    try {
   155→      const res = await fetch(`/api/dropdown-options?type=${type}`);
   156→      if (res.ok) setOptions(await res.json());
   157→    } catch { /* ignore */ }
   158→  }, [type]);
   159→
   160→  useEffect(() => { fetchOptions(); }, [fetchOptions]);
   161→
   162→  const handleAdd = async () => {
   163→    if (!newOptionValue.trim()) return;
   164→    setLoading(true);
   165→    try {
   166→      const res = await fetch("/api/dropdown-options", {
   167→        method: "POST",
   168→        headers: { "Content-Type": "application/json" },
   169→        body: JSON.stringify({ type, value: newOptionValue.trim() }),
   170→      });
   171→      if (res.ok) {
   172→        toast({ title: "Added", description: `"${newOptionValue.trim()}" added to ${label}` });
   173→        setNewOptionValue("");
   174→        await fetchOptions();
   175→      } else {
   176→        const err = await res.json();
   177→        toast({ title: "Error", description: err.error || "Failed to add", variant: "destructive" });
   178→      }
   179→    } catch {
   180→      toast({ title: "Error", description: "Failed to add option", variant: "destructive" });
   181→    } finally { setLoading(false); }
   182→  };
   183→
   184→  const handleUpdate = async (id: string) => {
   185→    if (!editingValue.trim()) return;
   186→    setLoading(true);
   187→    try {
   188→      const res = await fetch(`/api/dropdown-options/${id}`, {
   189→        method: "PUT",
   190→        headers: { "Content-Type": "application/json" },
   191→        body: JSON.stringify({ value: editingValue.trim() }),
   192→      });
   193→      if (res.ok) {
   194→        toast({ title: "Updated", description: "Option updated successfully" });
   195→        setEditingId(null);
   196→        setEditingValue("");
   197→        await fetchOptions();
   198→      } else {
   199→        toast({ title: "Error", description: "Failed to update", variant: "destructive" });
   200→      }
   201→    } catch {
   202→      toast({ title: "Error", description: "Failed to update", variant: "destructive" });
   203→    } finally { setLoading(false); }
   204→  };
   205→
   206→  const handleDelete = async (id: string, val: string) => {
   207→    setLoading(true);
   208→    try {
   209→      const res = await fetch(`/api/dropdown-options/${id}`, { method: "DELETE" });
   210→      if (res.ok) {
   211→        toast({ title: "Deleted", description: `"${val}" removed from ${label}` });
   212→        if (value === val) onChange("");
   213→        await fetchOptions();
   214→      }
   215→    } catch {
   216→      toast({ title: "Error", description: "Failed to delete", variant: "destructive" });
   217→    } finally { setLoading(false); }
   218→  };
   219→
   220→  return (
   221→    <div className="flex items-center gap-1">
   222→      <Select value={value} onValueChange={onChange} disabled={disabled}>
   223→        <SelectTrigger className="h-7 text-xs flex-1">
   224→          <SelectValue placeholder={`Select ${label}...`} />
   225→        </SelectTrigger>
   226→        <SelectContent>
   227→          {options.map((opt) => (
   228→            <SelectItem key={opt.id} value={opt.value}>{opt.value}</SelectItem>
   229→          ))}
   230→        </SelectContent>
   231→      </Select>
   232→      {!disabled && (
   233→        <Popover open={manageOpen} onOpenChange={setManageOpen}>
   234→          <PopoverTrigger asChild>
   235→            <Button variant="ghost" size="sm" className="h-7 w-7 p-0 shrink-0 text-slate-400 hover:text-slate-700" title="Manage options">
   236→              <Settings2 className="w-3.5 h-3.5" />
   237→            </Button>
   238→          </PopoverTrigger>
   239→          <PopoverContent className="w-72 p-0" align="end">
   240→            <div className="p-3 border-b bg-slate-50">
   241→              <h4 className="text-sm font-semibold text-slate-700">Manage {label}</h4>
   242→              <p className="text-xs text-slate-400 mt-0.5">Add, edit or delete options</p>
   243→            </div>
   244→            <div className="p-3 border-b">
   245→              <div className="flex gap-1.5">
   246→                <Input
   247→                  placeholder={`New ${label.toLowerCase()}...`}
   248→                  value={newOptionValue}
   249→                  onChange={(e) => setNewOptionValue(e.target.value)}
   250→                  className="h-7 text-xs"
   251→                  onKeyDown={(e) => e.key === "Enter" && handleAdd()}
   252→                />
   253→                <Button size="sm" onClick={handleAdd} disabled={loading || !newOptionValue.trim()} className="h-7 px-2 shrink-0 bg-slate-700 hover:bg-slate-600">
   254→                  <Plus className="w-3.5 h-3.5" />
   255→                </Button>
   256→              </div>
   257→            </div>
   258→            <div className="max-h-48 overflow-y-auto">
   259→              {options.length === 0 ? (
   260→                <p className="p-3 text-xs text-slate-400 text-center">No options yet</p>
   261→              ) : (
   262→                options.map((opt) => (
   263→                  <div key={opt.id} className="flex items-center gap-1 px-3 py-1.5 border-b last:border-b-0 hover:bg-slate-50 group">
   264→                    {editingId === opt.id ? (
   265→                      <>
   266→                        <Input
   267→                          value={editingValue}
   268→                          onChange={(e) => setEditingValue(e.target.value)}
   269→                          className="h-6 text-xs flex-1"
   270→                          onKeyDown={(e) => e.key === "Enter" && handleUpdate(opt.id)}
   271→                          autoFocus
   272→                        />
   273→                        <Button size="sm" onClick={() => handleUpdate(opt.id)} disabled={loading} className="h-6 w-6 p-0 shrink-0 text-green-600 hover:text-green-700">
   274→                          <Check className="w-3 h-3" />
   275→                        </Button>
   276→                        <Button size="sm" variant="ghost" onClick={() => { setEditingId(null); setEditingValue(""); }} className="h-6 w-6 p-0 shrink-0">
   277→                          <X className="w-3 h-3" />
   278→                        </Button>
   279→                      </>
   280→                    ) : (
   281→                      <>
   282→                        <span className="text-xs flex-1 truncate">{opt.value}</span>
   283→                        <Button
   284→                          size="sm" variant="ghost"
   285→                          onClick={() => { setEditingId(opt.id); setEditingValue(opt.value); }}
   286→                          className="h-6 w-6 p-0 shrink-0 opacity-0 group-hover:opacity-100 text-blue-500 hover:text-blue-700"
   287→                        >
   288→                          <Pencil className="w-3 h-3" />
   289→                        </Button>
   290→                        <Button
   291→                          size="sm" variant="ghost"
   292→                          onClick={() => handleDelete(opt.id, opt.value)}
   293→                          disabled={loading}
   294→                          className="h-6 w-6 p-0 shrink-0 opacity-0 group-hover:opacity-100 text-red-500 hover:text-red-700"
   295→                        >
   296→                          <Trash2 className="w-3 h-3" />
   297→                        </Button>
   298→                      </>
   299→                    )}
   300→                  </div>
   301→                ))
   302→              )}
   303→            </div>
   304→          </PopoverContent>
   305→        </Popover>
   306→      )}
   307→    </div>
   308→  );
   309→}

   311→// ==================== DatePickerWithAutoClose Component ====================
   312→function DatePickerWithAutoClose({
   313→  value,
   314→  onChange,
   315→}: {
   316→  value: string;
   317→  onChange: (val: string) => void;
   318→}) {
   319→  const [open, setOpen] = useState(false);
   320→
   321→  const parseDate = (dateStr: string): Date | undefined => {
   322→    if (!dateStr) return undefined;
   323→    const parts = dateStr.split("/");
   324→    if (parts.length === 3) {
   325→      return new Date(Number(parts[2]), Number(parts[1]) - 1, Number(parts[0]));
   326→    }
   327→    return undefined;
   328→  };
   329→
   330→  return (
   331→    <Popover open={open} onOpenChange={setOpen}>
   332→      <PopoverTrigger asChild>
   333→        <Button
   334→          variant="outline"
   335→          className="mt-1 w-full justify-start text-left font-normal h-9"
   336→        >
   337→          <CalendarIcon className="mr-2 h-4 w-4 text-slate-500" />
   338→          {value || "Pick a date"}
   339→        </Button>
   340→      </PopoverTrigger>
   341→      <PopoverContent className="w-auto p-0" align="start">
   342→        <Calendar
   343→          mode="single"
   344→          selected={parseDate(value)}
   345→          onSelect={(date) => {
   346→            if (date) {
   347→              onChange(format(date, "dd/MM/yyyy"));
   348→              setOpen(false);
   349→            }
   350→          }}
   351→          captionLayout="dropdown-buttons"
   352→          defaultMonth={parseDate(value) || new Date()}
   353→        />
   354→      </PopoverContent>
   355→    </Popover>
   356→  );
   357→}
