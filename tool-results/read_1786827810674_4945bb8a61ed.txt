   143→   143→// ==================== DropdownWithManage Component ====================
   144→   144→function DropdownWithManage({
   145→   145→  label,
   146→   146→  type,
   147→   147→  value,
   148→   148→  onChange,
   149→   149→  disabled,
   150→   150→}: {
   151→   151→  label: string;
   152→   152→  type: "department" | "branch" | "address";
   153→   153→  value: string;
   154→   154→  onChange: (val: string) => void;
   155→   155→  disabled: boolean;
   156→   156→  isAdmin: boolean;
   157→   157→}) {
   158→   158→  const [options, setOptions] = useState<DropdownOption[]>([]);
   159→   159→
   160→   160→  const fetchOptions = useCallback(async () => {
   161→   161→    try {
   162→   162→      const res = await fetch(`/api/dropdown-options?type=${type}`);
   163→   163→      if (res.ok) setOptions(await res.json());
   164→   164→    } catch { /* ignore */ }
   165→   165→  }, [type]);
   166→   166→
   167→   167→  useEffect(() => { fetchOptions(); }, [fetchOptions]);
   168→   168→
   169→   169→  return (
   170→   170→    <Select value={value} onValueChange={onChange} disabled={disabled}>
   171→   171→      <SelectTrigger className="h-7 text-xs w-full">

   173→   173→      </SelectTrigger>
   174→   174→      <SelectContent>
   175→   175→        {/* Show current value as fallback if it doesn't match any DB option */}
   176→   176→        {value && !options.some(opt => opt.value === value) && (
   177→   177→          <SelectItem key="__current__" value={value}>{value}</SelectItem>
   178→   178→        )}
   179→   179→        {options.map((opt) => (
   180→   180→          <SelectItem key={opt.id} value={opt.value}>{opt.value}</SelectItem>

   187→   187→// ==================== DatePickerWithAutoClose Component ====================
   188→   188→function DatePickerWithAutoClose({
   189→   189→  value,
   190→   190→  onChange,
   191→   191→}: {
   192→   192→  value: string;
   193→   193→  onChange: (val: string) => void;
   194→   194→}) {
   195→   195→  const [open, setOpen] = useState(false);
   196→   196→
   197→   197→  const parseDate = (dateStr: string): Date | undefined => {
   198→   198→    if (!dateStr) return undefined;
   199→   199→    const parts = dateStr.split("/");
   200→   200→    if (parts.length === 3) {
   201→   201→      return new Date(Number(parts[2]), Number(parts[1]) - 1, Number(parts[0]));
   202→   202→    }
   203→   203→    return undefined;
   204→   204→  };
   205→   205→
   206→   206→  return (
   207→   207→    <Popover open={open} onOpenChange={setOpen}>
   208→   208→      <PopoverTrigger asChild>
   209→   209→        <Button
   210→   210→          variant="outline"
   211→   211→          className="mt-1 w-full justify-start text-left font-normal h-9"
   212→   212→        >
   213→   213→          <CalendarIcon className="mr-2 h-4 w-4 text-slate-500" />
   214→   214→          {value || "Pick a date"}
   215→   215→        </Button>
   216→   216→      </PopoverTrigger>
   217→   217→      <PopoverContent className="w-auto p-0" align="start">
   218→   218→        <Calendar
   219→   219→          mode="single"
   220→   220→          selected={parseDate(value)}
   221→   221→          onSelect={(date) => {
   222→   222→            if (date) {
   223→   223→              onChange(format(date, "dd/MM/yyyy"));
   224→   224→              setOpen(false);
   225→   225→            }
   226→   226→          }}
   227→   227→          captionLayout="dropdown-buttons"
   228→   228→          defaultMonth={parseDate(value) || new Date()}
   229→   229→        />
   230→   230→      </PopoverContent>
   231→   231→    </Popover>
   232→   232→  );
   233→   233→}

   236→   236→function DropdownManagement({ type, label, icon }: { type: "department" | "branch" | "address"; label: string; icon: React.ReactNode }) {
   237→   237→  const [options, setOptions] = useState<DropdownOption[]>([]);
   238→   238→  const [newValue, setNewValue] = useState("");
   239→   239→  const [editingId, setEditingId] = useState<string | null>(null);
   240→   240→  const [editingValue, setEditingValue] = useState("");
   241→   241→  const [deleteId, setDeleteId] = useState<string | null>(null);
   242→   242→  const [deleteDialogOpen, setDeleteDialogOpen] = useState(false);
   243→   243→  const [loading, setLoading] = useState(false);
   244→   244→  const { toast } = useToast();
   245→   245→
   246→   246→  const fetchOptions = useCallback(async () => {
   247→   247→    try {
   248→   248→      const res = await fetch(`/api/dropdown-options?type=${type}`);
   249→   249→      if (res.ok) setOptions(await res.json());
   250→   250→    } catch { /* ignore */ }
   251→   251→  }, [type]);
   252→   252→
   253→   253→  useEffect(() => { fetchOptions(); }, [fetchOptions]);
   254→   254→
   255→   255→  const handleAdd = async () => {
   256→   256→    if (!newValue.trim()) return;
   257→   257→    setLoading(true);
   258→   258→    try {
   259→   259→      const res = await fetch("/api/dropdown-options", {
   260→   260→        method: "POST",
   261→   261→        headers: { "Content-Type": "application/json" },
   262→   262→        body: JSON.stringify({ type, value: newValue.trim() }),
   263→   263→      });
   264→   264→      if (res.ok) {
   265→   265→        toast({ title: "Added", description: `"${newValue.trim()}" added successfully` });
   266→   266→        setNewValue("");
   267→   267→        await fetchOptions();
   268→   268→      } else {
   269→   269→        const err = await res.json();
   270→   270→        toast({ title: "Error", description: err.error || "Failed to add", variant: "destructive" });
   271→   271→      }
   272→   272→    } catch {
   273→   273→      toast({ title: "Error", description: "Failed to add option", variant: "destructive" });
   274→   274→    } finally { setLoading(false); }
   275→   275→  };
   276→   276→
   277→   277→  const handleUpdate = async () => {
   278→   278→    if (!editingId || !editingValue.trim()) return;
   279→   279→    setLoading(true);
   280→   280→    try {
   281→   281→      const res = await fetch(`/api/dropdown-options/${editingId}`, {
   282→   282→        method: "PUT",
   283→   283→        headers: { "Content-Type": "application/json" },
   284→   284→        body: JSON.stringify({ value: editingValue.trim() }),
   285→   285→      });
   286→   286→      if (res.ok) {
   287→   287→        toast({ title: "Updated", description: "Option updated successfully" });
   288→   288→        setEditingId(null);
   289→   289→        setEditingValue("");
   290→   290→        await fetchOptions();
   291→   291→      } else {
   292→   292→        toast({ title: "Error", description: "Failed to update", variant: "destructive" });
   293→   293→      }
   294→   294→    } catch {
   295→   295→      toast({ title: "Error", description: "Failed to update", variant: "destructive" });
   296→   296→    } finally { setLoading(false); }
   297→   297→  };
   298→   298→
   299→   299→  const handleDelete = async () => {
   300→   300→    if (!deleteId) return;
   301→   301→    setLoading(true);
   302→   302→    try {
   303→   303→      const res = await fetch(`/api/dropdown-options/${deleteId}`, { method: "DELETE" });
   304→   304→      if (res.ok) {
   305→   305→        toast({ title: "Deleted", description: "Option removed successfully" });
   306→   306→        setDeleteId(null);
   307→   307→        setDeleteDialogOpen(false);
   308→   308→        await fetchOptions();
   309→   309→      }
   310→   310→    } catch {
   311→   311→      toast({ title: "Error", description: "Failed to delete", variant: "destructive" });
   312→   312→    } finally { setLoading(false); }
   313→   313→  };
   314→   314→
   315→   315→  return (
   316→   316→    <div className="space-y-6">
   317→   317→      {/* Header */}
   318→   318→      <div className="flex items-center gap-4">
   319→   319→        <div className="w-12 h-12 rounded-xl bg-slate-800 flex items-center justify-center text-white">
   320→   320→          {icon}
   321→   321→        </div>
   322→   322→        <div>
   323→   323→          <h2 className="text-xl font-bold text-slate-800">Manage {label}</h2>
   324→   324→          <p className="text-sm text-slate-500">Create, edit, and delete {label.toLowerCase()} options</p>
   325→   325→        </div>
   326→   326→      </div>
   327→   327→
   328→   328→      {/* Add New */}
   329→   329→      <Card className="shadow-sm">
   330→   330→        <CardContent className="p-4">
   331→   331→          <div className="flex items-center gap-3">
   332→   332→            <Input
   333→   333→              placeholder={`Enter new ${label.toLowerCase()} name...`}
   334→   334→              value={newValue}
   335→   335→              onChange={(e) => setNewValue(e.target.value)}
   336→   336→              onKeyDown={(e) => e.key === "Enter" && handleAdd()}
   337→   337→              className="flex-1 h-9"
   338→   338→            />
   339→   339→            <Button onClick={handleAdd} disabled={loading || !newValue.trim()} className="bg-slate-800 hover:bg-slate-700 gap-2 h-9">
   340→   340→              {loading ? <Loader2 className="w-4 h-4 animate-spin" /> : <Plus className="w-4 h-4" />} Add
   341→   341→            </Button>
   342→   342→          </div>
   343→   343→        </CardContent>
   344→   344→      </Card>
   345→   345→
   346→   346→      {/* Options List */}
   347→   347→      <Card className="shadow-sm">
   348→   348→        <div className="border-b bg-slate-50 px-4 py-3 flex items-center justify-between">
   349→   349→          <h3 className="text-sm font-semibold text-slate-700">{label} List ({options.length})</h3>
   350→   350→        </div>
   351→   351→        {options.length === 0 ? (
   352→   352→          <CardContent className="py-12 text-center">
   353→   353→            <Settings2 className="w-12 h-12 mx-auto text-slate-300 mb-3" />
   354→   354→            <p className="text-sm text-slate-500">No {label.toLowerCase()} options found. Add one above.</p>
   355→   355→          </CardContent>
   356→   356→        ) : (
   357→   357→          <div className="divide-y">
   358→   358→            {options.map((opt, idx) => (
   359→   359→              <div key={opt.id} className="px-4 py-3 flex items-center gap-3 hover:bg-slate-50 transition-colors">
   360→   360→                <span className="w-8 h-8 rounded-lg bg-slate-100 flex items-center justify-center text-xs font-semibold text-slate-600">{idx + 1}</span>
   361→   361→                {editingId === opt.id ? (
   362→   362→                  <div className="flex-1 flex items-center gap-2">
   363→   363→                    <Input
   364→   364→                      value={editingValue}
   365→   365→                      onChange={(e) => setEditingValue(e.target.value)}
   366→   366→                      onKeyDown={(e) => e.key === "Enter" && handleUpdate()}
   367→   367→                      className="h-8 flex-1"
   368→   368→                      autoFocus
   369→   369→                    />
   370→   370→                    <Button size="sm" onClick={handleUpdate} disabled={loading} className="gap-1 h-8 bg-emerald-600 hover:bg-emerald-500">
   371→   371→                      {loading ? <Loader2 className="w-3.5 h-3.5 animate-spin" /> : <Check className="w-3.5 h-3.5" />} Save
   372→   372→                    </Button>
   373→   373→                    <Button size="sm" variant="ghost" onClick={() => { setEditingId(null); setEditingValue(""); }} className="h-8">Cancel</Button>
   374→   374→                  </div>
   375→   375→                ) : (
   376→   376→                  <>
   377→   377→                    <span className="flex-1 text-sm font-medium text-slate-700">{opt.value}</span>
   378→   378→                    <Button variant="ghost" size="sm" onClick={() => { setEditingId(opt.id); setEditingValue(opt.value); }} className="h-8 w-8 p-0" title="Edit">
   379→   379→                      <Edit3 className="w-3.5 h-3.5 text-blue-600" />
   380→   380→                    </Button>
   381→   381→                    <Button variant="ghost" size="sm" onClick={() => { setDeleteId(opt.id); setDeleteDialogOpen(true); }} className="h-8 w-8 p-0" title="Delete">
   382→   382→                      <Trash2 className="w-3.5 h-3.5 text-red-500" />
   383→   383→                    </Button>
   384→   384→                  </>
   385→   385→                )}
   386→   386→              </div>
   387→   387→            ))}
   388→   388→          </div>
   389→   389→        )}
   390→   390→      </Card>
   391→   391→
   392→   392→      {/* Delete Confirmation Dialog */}
   393→   393→      <Dialog open={deleteDialogOpen} onOpenChange={setDeleteDialogOpen}>
   394→   394→        <DialogContent>
   395→   395→          <DialogHeader>
   396→   396→            <DialogTitle>Delete {label}</DialogTitle>
   397→   397→            <DialogDescription>Are you sure you want to delete this {label.toLowerCase()} option? This action cannot be undone.</DialogDescription>
   398→   398→          </DialogHeader>
   399→   399→          <DialogFooter>
   400→   400→            <Button variant="outline" onClick={() => setDeleteDialogOpen(false)}>Cancel</Button>
   401→   401→            <Button variant="destructive" onClick={handleDelete} disabled={loading}>
   402→   402→              {loading ? <Loader2 className="w-4 h-4 animate-spin mr-2" /> : null} Delete
   403→   403→            </Button>
   404→   404→          </DialogFooter>
   405→   405→        </DialogContent>
   406→   406→      </Dialog>
   407→   407→    </div>
   408→   408→  );
   409→   409→}
